! Program: Sum from 1 to N
! Author: 劉祐良 (413210351)
! Description: This program calculates the sum of integers from 1 to a user-specified positive integer N.
! Formula: sum = N*(N+1)/2, but implemented with a loop for educational purposes.
program sum_1_to_n
  implicit none
  integer :: n, i, total
  
  ! Prompt the user to enter a positive integer N
  write(*,*) 'Enter a positive integer N:'
  read(*,*) n
  
  ! Initialize total to zero
  total = 0
  
  ! Loop from 1 to N and accumulate the sum
  do i = 1, n
    total = total + i
  end do
  
  ! Output the calculated sum
  write(*,*) 'The sum from 1 to', n, 'is', total
end program sum_1_to_n
